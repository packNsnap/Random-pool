C. Scheduler logic (pseudo)
Runs every morning.
def scheduler_tick():
    today = date.today()
    current_quarter = get_current_quarter(today)

    for client in get_active_clients():
        # 1) Quarterly roster reminder logic
        if needs_roster_reminder(client, current_quarter, today):
            send_roster_email(client, template_type="roster_request", quarter=current_quarter)

        # 2) Follow up logic
        if needs_follow_up(client, current_quarter, today):
            send_roster_email(client, template_type="follow_up", quarter=current_quarter)

        # 3) Biweekly progress update
        if needs_progress_update(client, today):
            send_progress_email(client)

Each send_* function:


Renders subject/body from template and variables.


Sends email via SMTP or Gmail API.


Logs to email_logs.


Updates clients table (status and dates).


D. Minimal FastAPI skeleton
I’m not dumping a massive codebase here, but this is what your main.py shape looks like:
from fastapi import FastAPI, Request, Depends, Form
from fastapi.responses import HTMLResponse, RedirectResponse
from fastapi.templating import Jinja2Templates
from apscheduler.schedulers.background import BackgroundScheduler
from datetime import datetime

app = FastAPI()
templates = Jinja2Templates(directory="templates")

scheduler = BackgroundScheduler()
scheduler.add_job(func=scheduler_tick, trigger="interval", hours=24)
scheduler.start()

@app.get("/", response_class=HTMLResponse)
async def dashboard(request: Request, user=Depends(require_login)):
    clients = get_clients_with_status()
    return templates.TemplateResponse("dashboard.html", {"request": request, "user": user, "clients": clients})

@app.get("/clients", response_class=HTMLResponse)
async def list_clients(request: Request, user=Depends(require_login)):
    clients = get_all_clients()
    return templates.TemplateResponse("clients.html", {"request": request, "clients": clients})

@app.post("/clients/{client_id}/send_roster_request")
async def manual_send_roster(client_id: int, user=Depends(require_login)):
    client = get_client(client_id)
    send_roster_email(client, template_type="roster_request", quarter=get_current_quarter(datetime.utcnow().date()))
    return RedirectResponse(url=f"/clients/{client_id}", status_code=303)
